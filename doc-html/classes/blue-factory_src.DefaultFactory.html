    Class DefaultFactory

    A default implementation of AbstractFactory.

    Example

    const factory = new DefaultFactory();
    factory.setValue( 'foo', 123 );
    factory.getValue( 'foo' );
    //=> 123
    factory.hasValue( 'foo' );
    //=> true
    factory.deleteValue( 'foo' );
    factory.hasValue( 'foo' );
    //=> false
