/**
 * @function module:legacy.util.object.every
 * @see {@link http://moutjs.com/docs/latest/object.html#every}
 */
export const every = (obj, callback, thisObj) => {
    callback = makeIterator(callback, thisObj);
    let result = true;
    for (let key in obj) {
        if (hasOwnProperty.call(obj, key)) {
            if (!callback(obj[key], key, obj)) {
                result = false;
                break;
            }
        }
    }
    return result;
};

 * @function module:legacy.util.object.mixIn

export const mixIn = (target, ...objects) => {
    let index = 0;
    const length = objects.length;
    for (; index < length; index++) {
        const object = objects[index];
        if (isNotNil(object)) {
            forEachObjIndexed(function (val, key, obj) {
                if (hasOwnProperty.call(obj, key)) {
                    target[key] = val;
                }
            }, object);
        }
    }
    return target;
};

export const set = (obj, prop, val) => {
    const path = splitOnDot(prop), length = path.length;
    if (length > 1) {
        let context = obj;
        forEachIndex((key, index) => {
            // @ts-ignore: ¯\_(ツ)_/¯
            when(propIsNil(key), (ctx) => (ctx[key] = index === length - 1 ? val : {}))(context);
            context = context[key];
        }, path);
    }
    else {
        obj[prop] = val;
    }
    return obj;
};

/**
 * Delete object property if existent and returns a boolean indicating succes. It will also return true if property doesn't exist.
 * @function module:legacy.util.object.unset
 * @param {external:Object} obj
 * @param {external:string} prop
 * @returns {external:boolean}
 * @example
 * let lorem = { ipsum: { dolor: { sit: 'amet' } } };
 * unset( lorem, 'ipsum.dolor.sit' ); // true
 * console.log( lorem.ipsum.dolor );  // {}
 * unset( lorem, 'foo.bar' );         // true
 * @see {@link http://moutjs.com/docs/latest/object.html#unset}
 */
export const unset = (obj, prop) => {
    const path = splitOnDot(prop);
    return ifElse(hasPath(path), (val) => {
        const last = path.pop();
        while (prop = path.shift()) {
            val = val[prop];
        }
        return Reflect.deleteProperty(val, last);
    }, T)(obj);
};
