/**
 * @function module:core~addConditionalEventListener
 * @param {module:context.EventEmitter} emitter The emitter on which the event would be emitted.
 * @param {module:context.EventType} type The event type.
 * @param {module:context.EventListener} listener The event callback.
 */
function addConditionalEventListener(emitter, type, listener) {
    const conditionalListener = function () {
        const done = listener.apply(emitter, arguments);
        if (done === true) {
            removeEventListener(emitter, type, conditionalListener);
        }
    };
    conditionalListener.listener = listener.listener || listener;
    addEventListener(emitter, type, conditionalListener, NaN);
}
/**
 * @function module:core~addEventHandler
 * @param {module:context.EventEmitter} emitter The emitter on which the event would be emitted.
 * @param {module:context.EventType} type The event type.
 * @param {module:context.EventHandler} handler The event handler.
 */
function addEventHandler(emitter, type, handler) {
    let index, length;
    // List of listeners
    if (isArray(handler)) {
        index = 0;
        length = handler.length;
        for (; index < length; index += 1) {
            addEventListener(emitter, type, handler[index], NaN);
        }
        // Single listener
    }
    else {
        addEventListener(emitter, type, handler, NaN);
    }
}
/**
 * @function module:core~addEventHandlers
 * @param {module:context.EventEmitter} emitter The emitter on which the event would be emitted.
 * @param {module:context.EventHandlers} mappings The event mapping.
 */
function addEventHandlers(emitter, mappings, prefixed = false) {
    let handler, parts, type;
    /**
     * Intentionally using a for-in loop without a hasOwnProperty check in order to
     * check the prototype chain for mappings
     */
    for (const key in mappings) {
        if (!prefixed) {
            type = key;
        }
        else if (parts = onMatcher.exec(key)) {
            // Example: if key = "onChange" then parts = [ "onChange", "Change" ] and type = "change"
            const capitalizedType = parts[1];
            type = `${capitalizedType[0].toLowerCase()}${capitalizedType.slice(1)}`;
        }
        if (type) {
            handler = mappings[type];
            isEventHandler(handler) && addEventHandler(emitter, type, handler);
        }
    }
}
/**
 * @function module:core~addEventListener
 * @param {module:context.EventEmitter} emitter The emitter on which the event would be emitted.
 * @param {module:context.EventType} type The event type.
 * @param {module:context.EventListener} listener The event callback.
 * @param {external:number} index
 */
function addEventListener(emitter, type, listener, index) {
    // Define the event registry if it does not exist
    defineEventsProperty(emitter, createNull());
    const _events = emitter[$events];
    if (_events[':on']) {
        emitEvent(emitter, ':on', [type, hasListener(listener) ? listener.listener : listener], false);
        // Emitting "on" may have changed the registry.
        _events[':on'] = emitter[$events][':on'];
    }
    // Single listener
    if (!_events[type]) {
        _events[type] = listener;
        // Multiple listeners
    }
    else if (isArray(_events[type])) {
        switch (isNaN(index) || index) {
            case true:
                _events[type].push(listener);
                break;
            case 0:
                _events[type].unshift(listener);
                break;
            default:
                _events[type].splice(index, 0, listener);
                break;
        }
        // Transition from single to multiple listeners
    }
    else {
        _events[type] = index === 0 ?
            [listener, _events[type]] :
            [_events[type], listener];
    }
    emitter[$events] = _events;
}
/**
 * @function module:core~addFiniteEventListener
 * @param {module:context.EventEmitter} emitter The emitter on which the event would be emitted.
 * @param {module:context.EventType} type The event type.
 * @param {external:number} times The number times the listener will be executed before being removed.
 * @param {module:context.EventListener} listener The event callback.
 */
function addFiniteEventListener(emitter, type, times, listener) {
    const finiteListener = function () {
        listener.apply(this, arguments);
        return --times === 0;
    };
    finiteListener.listener = listener.listener || listener;
    addConditionalEventListener(emitter, type, finiteListener);
}
/**
 * @function module:core~defineEventsProperty
 * @param {*} target The target on which the property will be created.
 * @param {module:context.EventHandlers} value The initial value of the event map.
 */
function defineEventsProperty(target, value) {
    const hasEvents = hasOwnProperty.call(target, $events), targetPrototype = getPrototypeOf(target);
    if (!hasEvents || (targetPrototype && target[$events] === targetPrototype[$events])) {
        defineProperty(target, $events, {
            value: value,
            configurable: true,
            enumerable: false,
            writable: true
        });
    }
}
/**
 * @function module:core~emitAllEvents
 * @param {module:context.EventEmitter} emitter The emitter on which the event `type` will be emitted.
 * @param {module:context.EventType} type The event type.
 * @param {external:Array} data The data to be passed with the event.
 * @return {external:boolean} Whether or not a listener for the given event type was executed.
 * @throws {external:Error} If `type` is `error` and no listeners are subscribed.
 */
function emitAllEvents(emitter, type, data) {
    let executed = false, index = typeof type === 'string' ? lastIndexOfColon(type) : -1;
    // Namespaced event, e.g. Emit "foo:bar:qux", then "foo:bar"
    while (index > 0) {
        executed = (type && emitEvent(emitter, type, data, false)) || executed;
        type = type.substring(0, index);
        index = lastIndexOfColon(type);
    }
    // Emit single event or the namespaced event root, e.g. "foo", ":bar", Symbol( "@@qux" )
    executed = (type && emitEvent(emitter, type, data, type !== $every)) || executed;
    return executed;
}
/**
 * @function module:core~emitErrors
 * @param {module:context.EventEmitter} emitter The emitter on which the `errors` will be emitted.
 * @param {external:Array<Error>} errors The array of errors to be emitted.
 */
function emitErrors(emitter, errors) {
    const length = errors.length;
    for (let index = 0; index < length; index += 1) {
        emitEvent(emitter, 'error', [errors[index]], false);
    }
}
/**
 * @function module:core~emitEvent
 * @param {module:context.EventEmitter} emitter The emitter on which the event `type` will be emitted.
 * @param {module:context.EventType} type The event type.
 * @param {external:Array} data The data to be passed with the event.
 * @param {external:boolean} emitEvery Whether or not listeners for all types will be executed.
 * @return {external:boolean} Whether or not a listener for the given event type was executed.
 * @throws {external:Error} If `type` is `error` and no listeners are subscribed.
 */
function emitEvent(emitter, type, data, emitEvery) {
    const _events = emitter[$events];
    let executed = false, listener;
    if (isDefined(_events)) {
        if (type === 'error' && !_events.error) {
            if (data[0] instanceof Error) {
                throw data[0];
            }
            else {
                throw new Error('Uncaught, unspecified "error" event.');
            }
        }
        // Execute listeners for the given type of event
        listener = _events[type];
        if (isDefined(listener)) {
            executeHandler(listener, data, emitter);
            executed = true;
        }
        // Execute listeners listening for all types of events
        if (emitEvery) {
            listener = _events[$every];
            if (isDefined(listener)) {
                executeHandler(listener, data, emitter);
                executed = true;
            }
        }
    }
    return executed;
}
/**
 * @function module:core~eventListenerEquals
 * @param {module:context.EventListener} l1
 * @param {module:context.EventListener} l2

const eventListenerEquals = (l1, l2) => either(equals(l2), propEq('listener', l2))(l1);
/**
 * Executes a listener using the internal `execute*` functions based on the number of arguments.
 * @function module:core~executeHandler
 * @param {module:context.EventHandler} handler
 * @param {external:Array} data
 * @param {*} scope
 */
function executeHandler(handler, data, scope) {
    const isFn = isFunction(handler);
    switch (data.length) {
        case 0:
            handleEmpty(handler, isFn, scope);
            break;
        case 1:
            handleOne(handler, isFn, scope, data[0]);
            break;
        case 2:
            handleTwo(handler, isFn, scope, data[0], data[1]);
            break;
        case 3:
            handleThree(handler, isFn, scope, data[0], data[1], data[2]);
            break;
        default:
            handleMany(handler, isFn, scope, data);
            break;
    }
}

/**
 * Execute a listener with no arguments.
 * @function module:core~handleMany
 * @param {module:context.EventHandler} handler One or more {@link EventListener|listeners} that will be executed on the `emitter`.
 * @param {external:boolean} isFunction Whether or not the `handler` is a {@link Function|function}.
 * @param {module:context.EventEmitter} emitter The emitter.
 */
function handleEmpty(handler, isFunction, emitter) {
    const errors = [];
    if (isFunction) {
        try {
            handler.call(emitter);
        }
        catch (error) {
            errors.push(error);
        }
    }
    else {
        const length = handler.length, listeners = handler.slice();
        let index = 0;
        for (; index < length; index += 1) {
            try {
                listeners[index].call(emitter);
            }
            catch (error) {
                errors.push(error);
            }
        }
    }
    if (errors.length) {
        emitErrors(emitter, errors);
    }
}

/**
 * Remove all listeners, or those for the specified event `type`.
 * @function module:core~removeAllEventListeners
 * @param {module:context.EventEmitter} emitter The emitter from which listeners will be removed.
 * @param {module:context.EventType} [type] The event type.
 */
function removeAllEventListeners(emitter, type) {
    let handler;
    if (emitter[$events]) {
        // With no "off" listeners, clearing can be simplified
        if (!emitter[$events][':off']) {
            if (isEventType(type)) {
                emitter[$events] = createNull();
            }
            else if (emitter[$events][type]) {
                deleteProperty(emitter[$events], type);
            }
            // Clear all listeners
        }
        else if (isEventType(type)) {
            const types = getEventTypes(emitter), length = types.length;
            // Avoid removing "off" listeners until all other types have been removed
            for (let index = 0; index < length; index += 1) {
                if (types[index] === ':off') {
                    continue;
                }
                removeAllEventListeners(emitter, types[index]);
            }
            // Manually clear "off"
            removeAllEventListeners(emitter, ':off');
            emitter[$events] = createNull();
        }
        else {
            handler = emitter[$events][type];
            removeEventHandler(emitter, type, handler);
            deleteProperty(emitter[$events], type);
        }
    }
}
/**
 * @function module:core~removeEventHandler
 * @param {module:context.EventEmitter} emitter The emitter from which the handler would be removed.
 * @param {module:context.EventType} type The event type.
 * @param {module:context.EventHandler} handler The event callback.
 */
function removeEventHandler(emitter, type, handler) {
    if (typeof handler === 'function') {
        removeEventListener(emitter, type, handler);
    }
    else if (isArray(handler)) {
        let index = handler.length;
        while (index--) {
            removeEventListener(emitter, type, handler[index]);
        }
    }
}
/**
 * @function module:core~removeEventListener
 * @param {module:context.EventEmitter} emitter The emitter from which the listener would be removed.
 * @param {module:context.EventType} type The event type.
 * @param {module:context.EventListener} listener The event callback.
 */
function removeEventListener(emitter, type, listener) {
    const handler = emitter[$events][type];
    if (eventListenerEquals(handler, listener)) {
        deleteProperty(emitter[$events], type);
        if (emitter[$events][':off']) {
            emitEvent(emitter, ':off', [type, listener], false);
        }
    }
    else if (isArray(handler)) {
        let index = -1;
        for (let i = handler.length; i-- > 0;) {
            if (eventListenerEquals(handler[i], listener)) {
                index = i;
                break;
            }
        }
        if (index > -1) {
            if (handler.length === 1) {
                handler.length = 0;
                deleteProperty(emitter[$events], type);
            }
            else {
                spliceList(handler, index);
            }
            if (emitter[$events][':off']) {
                emitEvent(emitter, ':off', [type, listener], false);
            }
        }
    }
}
/**
 * Faster than `Array.prototype.splice`
 * @function module:core~spliceList
 * @param {external:Array} list
 * @param {external:number} index
 */
function spliceList(list, index) {
    const length = list.length;
    let i = index, j = i + 1;
    for (; j < length; i += 1, j += 1) {
        list[i] = list[j];
    }
    list.pop();
}
/**
 * @function module:core~tickAllEvents
 * @param {module:context.EventEmitter} emitter The emitter on which the event `type` will be asynchronously emitted.
 * @param {module:context.EventType} type The event type.
 * @param {external:Array} data The data to be passed with the event.
 * @return {external:Promise} A promise which *resolves* if the event had listeners, *rejects* otherwise.
 */
async function tickAllEvents(emitter, type, data) {
    return new Promise(function (resolve, reject) {
        queueMicrotask(() => emitAllEvents(emitter, type, data) ? resolve() : reject());
    });
}
/**
 * @mixin module:context.AsAsyncEventEmitter
 * @implements module:context.AsyncEventEmitter
 * @returns {module:core.Constructable<module:context.AsyncEventEmitter>}
 */
export const AsAsyncEventEmitter = (superclass) => class extends superclass {
    /**
     * Asynchronously emits specified event `type` with the supplied arguments. The listeners will still be synchronously executed in the specified order.
     *
     * The `type` can be namespaced using `:`, which will result in multiple events being triggered in succession. Listeners can be associated with the fully namespaced `type` or a subset of the `type`.
     *
     * Returns {@link external:Promise|promise} which *resolves* if the event had listeners, *rejects* otherwise.
     * @function module.context.AsAsyncEventEmitter#tick
     * @param {module:context.EventType} type The event type.
     * @param {...*} [data] The data passed into the listeners.
     * @return {external:Promise} A promise which *resolves* if the event had listeners, *rejects* otherwise.
     * @since 2.0.0
     * @example <caption>Asynchronously emitting an event</caption>
     * const greeter = new Emitter();
     * greeter.on( 'hello', () => console.log( 'Hello!' ) );
     * greeter.tick( 'hello' ).then( ( heard ) => console.log( 'hello heard? ', heard ) );
     * greeter.tick( 'goodbye' ).then( ( heard ) => console.log( 'goodbye heard? ', heard ) );
     * // Hello!
     * // hello heard? true
     * // goodbye heard? false
     */
    async tick(type, ...data) {
        return tickAllEvents(this, type, data);
    }
};
/**
 * @mixin module:context.AsConditionalEventEmitter
 * @implements module:context.ConditionalEventEmitter
 * @returns {module:core.Constructable<module:context.ConditionalEventEmitter>}
 */
export const AsConditionalEventEmitter = (superclass) => class extends superclass {
    /**
     * Adds a *many time* listener for the specified event `type`. If no `type` is given the listener will be triggered any event `type`. After the listener is invoked the specified number of `times`, it is removed.
     * No checks are made to see if the `listener` has already been added. Multiple calls passing the same combination `type` and `listener` will result in the `listener` being added multiple times.
     * @function module:context.AsConditionalEventEmitter#many
     * @param {module:context.EventType} type The event type.
     * @param {external:number} times The number times the listener will be executed before being removed.
     * @param {module:context.EventListener} listener The event callback.
     * @return {module:context.EventEmitter} The emitter.
     * @since 1.0.0
     * @example <caption>Listen to any event type a set number of times</caption>
     * const greeter = new Emitter();
     * greeter.many( 2, ( name ) => console.log( `Greeted ${ name }` ) );
     * greeter.emtest( 'hello', 'Jeff' );    // 1
     * // Greeted Jeff
     * greeter.emtest( 'hi', 'Terry' );      // 2
     * // Greeted Terry
     * greeter.emtest( 'yo', 'Steve' );      // 3
     * @example <caption>Listen to the specified event type a set number of times</caption>
     * const greeter = new Emitter();
     * greeter.many( 'hello', 2, ( name ) => console.log( `Hello, ${ name }!` ) );
     * greeter.emtest( 'hello', 'Jeff' );    // 1
     * // Hello, Jeff!
     * greeter.emtest( 'hello', 'Terry' );   // 2
     * // Hello, Terry!
     * greeter.emtest( 'hello', 'Steve' );   // 3
     */
    many(type = $every, times, listener) {
        // Shift arguments if type is not provided
        if (typeof type === 'number' && typeof times === 'function' && typeof listener === 'undefined') {
            listener = times;
            times = type;
            type = $every;
        }
        if (!isPositiveNumber(times)) {
            throw new TypeError('times must be a positive number');
        }
        if (typeof listener !== 'function') {
            throw new TypeError('listener must be a function');
        }
        addFiniteEventListener(this, type, times, listener);
        return this;
    }
    /**
     * @function module:context.AsConditionalEventEmitter#once
     * @param {module:context.EventType} [type] The event type.
     * @param {module:context.EventListener} listener The event callback.
     * @return {module:context.EventEmitter} The emitter.
     * @since 1.0.0
     * @emits {:on} when listeners are added
     * const greeter = new Emitter();
     * greeter.once( () => console.log( 'Greeted' ) );
     * greeter.emtest( 'hello' );
     * // Greeted
     * greeter.emtest( 'goodbye' );
     * @example <caption>Listen once to all event types</caption>
     * const greeter = new Emitter();
     * greeter.once( 'hello', ( name ) => console.log( `Hello, ${ name }!` ) );
     * greeter.emtest( 'hello', 'World' );
     * // Hello, World!
     * greeter.emtest( 'hello', 'World' );

    once(type = $every, listener) {
        // Shift arguments if type is not provided
        if (typeof type === 'function' && typeof listener === 'undefined') {
            listener = type;
            type = $every;
        }
        if (typeof listener !== 'function') {
            throw new TypeError('listener must be a function');
        }
        addFiniteEventListener(this, type, 1, listener);
        return this;
    }
    /**
     * Adds a listeners for the specified event `type` that will be triggered *until* the `listener` returns `true`. If no `type` is given the listener will be triggered any event `type`.
     *
     * No checks are made to see if the `listener` has already been added. Multiple calls passing the same combination `type` and `listener` will result in the `listener` being added multiple times.
     * @function module:context.AsConditionalEventEmitter#until
     * @param {module:context.EventType} [type] The event type.
     * @param {module:context.EventListener} listener The event callback.
     * @return {module:context.EventEmitter} The emitter.
     * @since 1.2.0
     * @example
     * const greeter = new Emitter();
     * greeter.until( function( name ){
     *  console.log( `Greeted ${ name }` );
     *  return name === 'Terry';
     * } );
     * greeter.emtest( 'hello', 'Jeff' );
     * // Greeted Jeff
     * greeter.emtest( 'goodbye', 'Terry' );
     * // Greeted Terry
     * greeter.emtest( 'hi', 'Aaron' );
     * @example
     * const greeter = new Emitter();
     * greeter.until( 'hello', function( name ){
     *  console.log( `Hello, ${ name }!` );
     *  return name === 'World';
     * } );
     * greeter.emtest( 'hello', 'Jeff' );
     * // Hello, Jeff!
     * greeter.emtest( 'hello', 'World' );
     * // Hello, World!
     * greeter.emtest( 'hello', 'Mark' );
     */
    until(type = $every, listener) {
        // Shift arguments if type is not provided
        if (typeof type === 'function' && typeof listener === 'undefined') {
            listener = type;
            type = $every;
        }
        if (typeof listener !== 'function') {
            throw new TypeError('listener must be a function');
        }
        addConditionalEventListener(this, type, listener);
        return this;
    }
};
export const asEventEmitter = function () {
    this.emit = function (type, ...data) {
        return emitAllEvents(this, type, data);
    };
    this.on = function (type = $every, listener) {
        if (typeof listener === 'undefined') {
            // Type not provided, fall back to "$every"
            if (typeof type === 'function') {
                listener = type;
                type = $every;
                // Plain object of event bindings
            }
            else if (typeof type === 'object') {
                addEventHandlers(this, type, false);
                return this;
            }
        }
        addEventListener(this, type, listener, NaN);
        return this;
    };
    this.off = function (type, listener) {
        // Shift arguments if type is not provided
        if (typeof type === 'function' && typeof listener === 'undefined') {
            listener = type;
            type = $every;
        }
        if (typeof listener !== 'function') {
            throw new TypeError('listener must be a function');
        }
        if (!this[$events] || !this[$events][type]) {
            return this;
        }
        removeEventListener(this, type, listener);
        return this;
    };
};

    /**
     * Execute the listeners for the specified event `type` with the supplied arguments.
     *
     * The `type` can be namespaced using `:`, which will result in multiple events being triggered in succession. Listeners can be associated with the fully namespaced `type` or a subset of the `type`.
     *
     * Returns `true` if the event had listeners, `false` otherwise.
     * @function module:context.AsEventEmitter#emit
     * @param {module:context.EventType} type The event type.
     * @param {...*} [data] The data passed into the listeners.

     * const greeter = new Emitter();
     * greeter.on( 'hello', () => console.log( 'Hello!' ) );
     * greeter.emtest( 'hello' );     // true
     * // Hello!
     * greeter.emtest( 'goodbye' );   // false
     * @example <caption>Emitting an event with data</caption>
     * const greeter = new Emitter();
     * greeter.on( 'hello', ( name ) => console.log( `Hello, ${ name }!` ) );
     * greeter.emtest( 'hello', 'World' );
     * // Hello, World!
     * @example <caption>Emitting a namespaced event</caption>
     * const greeter = new Emitter();
     * greeter.on( 'greeting:hello', ( name ) => console.log( `Hello, ${ name }!` ) );
     * greeter.on( 'greeting:hi', ( name ) => console.log( `Hi, ${ name }!` ) );
     * greeter.on( 'greeting', ( name ) => console.log( `${ name } was greeted.` );
     *
     * // This event will not be triggered by emitting "greeting:hello"
     * greeter.on( 'hello', ( name ) => console.log( `Hello again, ${ name }` );
     *

     * // Mark was greeted.
     *
     * greeter.emtest( 'greeting:hello', 'Jeff' );
     * // Hello, Jeff!
     * // Jeff was greeted.

        return emitAllEvents(this, type, data);
    }
    /**
     * @function module:context.AsEventEmitter#eventTypes
     * @return {external:Array<module:context.EventType>} The list of event types registered to the emitter.
     * @since 2.0.0
     * @example
     * const greeter = new Emitter();

     * greeter.on( 'hi', () => console.log( `Hi` ) );
     *
     * console.log( greeter.eventTypes() );
     * // [ 'hello', 'hi' ]

        return getEventTypes(this);
    }
    /**
     * @function module:context.AsEventEmitter#listenerCount
     * @param {module:context.EventType} type The event type.
     * @return {external:number} The number of listeners for that event type within the given emitter.
     * @since 1.0.0
     * @example
     * const greeter = new Emitter();
     * greeter.on( 'hello', () => console.log( 'Hello!' ) );
     * console.log( greeter.listenerCount( 'hello' ) );
     * // 1
     * console.log( greeter.listenerCount( 'goodbye' ) );
     * // 0
     */
    listenerCount(type) {
        let count;
        // Empty
        if (!this[$events] || !this[$events][type]) {
            count = 0;
            // Function
        }
        else if (typeof this[$events][type] === 'function') {
            count = 1;
            // Array
        }
        else {
            count = this[$events][type].length;
        }
        return count;
    }
    /**
     * @function module:context.AsEventEmitter#listeners
     * @param {module:context.EventType} type The event type.
     * @return {external:number} The number of listeners for that event type within the given emitter.
     * @since 1.0.0
     * @example
     * const hello = function(){
     *  console.log( 'Hello!' );
     * },
     *  greeter = new Emitter();
     *
     * greeter.on( 'hello', hello );
     * greeter.emtest( 'hello' );
     * // Hello!
     *
     * console.log( greeter.listeners( 'hello' )[ 0 ] === hello );
     * // true
     */
    listeners(type) {
        let listeners;
        if (!this[$events] || !this[$events][type]) {
            listeners = [];
        }
        else {
            const handler = this[$events][type];
            if (typeof handler === 'undefined') {
                listeners = [];
            }
            else if (typeof handler === 'function') {
                listeners = [handler];
            }
            else {
                listeners = handler.slice();
            }
        }
        return listeners;
    }
    /**
     * Adds a listener for the specified event `type`. If no `type` is given the listener will be triggered any event `type`.
     *
     * No checks are made to see if the `listener` has already been added. Multiple calls passing the same combination `type` and `listener` will result in the `listener` being added multiple times.
     * @function module:context.AsEventEmitter#on
     * @param {module:context.EventType} [type] The event type.
     * @param {module:context.EventListener} listener The event callback.
     * @return {module:context.EventEmitter} The emitter.
     * @since 1.0.0
     * @emits {:on} when listeners are added
     * @example <caption>Listen to all event types</caption>
     * const greeter = new Emitter();
     * greeter.on( () => console.log( 'Greeted' ) );
     * greeter.emtest( 'hello' );
     * // Greeted
     * greeter.emtest( 'goodbye' );
     * // Greeted
     * @example <caption>Listener to a specified event type</caption>
     * const greeter = new Emitter();
     * greeter.on( 'hello', ( name ) => console.log( `Hello, ${ name }!` ) );
     * greeter.emtest( 'hello', 'World' );
     * // Hello, World!
     * greeter.emtest( 'hi', 'World' );
     */
    on(type = $every, listener) {
        if (typeof listener === 'undefined') {
            // Type not provided, fall back to "$every"
            if (typeof type === 'function') {
                listener = type;
                type = $every;
                // Plain object of event bindings
            }
            else if (typeof type === 'object') {
                addEventHandlers(this, type, false);
                return this;
            }
        }
        addEventListener(this, type, listener, NaN);
        return this;
    }
    /**
     * Removes the `listener` for the specified event `type`. If no `type` is given it is assumed the `listener` is not associated with a specific `type`.
     *
     * If any single listener has been added multiple times for the specified `type`, then `emitter.off()` must be called multiple times to remove each instance.
     * @function module:context.AsEventEmitter#off
     * @param {module:context.EventType} type The event type.
     * @param {module:context.EventListener} listener The event callback.
     * @return {module:context.EventEmitter} The emitter.
     * @since 1.0.0
     * @emits {:off} when listeners are removed
     * @example <caption>Remove a listener from any event type</caption>
     * function greet( name ){
     *  console.log( `Greetings, ${ name }!` );
     * }
     *
     * const greeter = new Emitter();
     * greeter.on( greet );
     * greeter.emtest( 'hello' 'Jeff' );
     * // Greetings, Jeff!
     * greeter.emtest( 'hi' 'Jeff' );
     * // Greetings, Jeff!
     * greeter.off( greet );
     * greeter.emtest( 'yo', 'Jeff' );
     * @example <caption>Remove a listener from a specified event type</caption>
     * function hello( name ){
     *  console.log( `Hello, ${ name }!` );
     * }
     *
     * const greeter = new Emitter();
     * greeter.on( 'hello', hello );
     * greeter.emtest( 'hello', 'Jeff' );
     * // Hello, Jeff!
     * greeter.off( 'hello', hello );
     * greeter.emtest( 'hello', 'Jeff' );
     */
    off(type, listener) {
        // Shift arguments if type is not provided
        if (typeof type === 'function' && typeof listener === 'undefined') {
            listener = type;
            type = $every;
        }
        if (typeof listener !== 'function') {
            throw new TypeError('listener must be a function');
        }
        if (!this[$events] || !this[$events][type]) {
            return this;
        }
        removeEventListener(this, type, listener);
        return this;
    }
};

    prependOnceListener(type, listener) {
        let times = 1;
        const finiteListener = function () {
            listener.apply(this, arguments);
            return --times === 0;

        finiteListener.listener = listener;
        const conditionalListener = function () {
            const done = finiteListener.apply(this, arguments);
            if (done === true) {
                removeEventListener(this, type, conditionalListener);
            }
        };
        conditionalListener.listener = listener.listener || listener;
        addEventListener(this, type, conditionalListener, 0);
        return this;
    }

        removeAllEventListeners(this, type);
        return this;
    }
    removeListener(type, listener) {
        return this.off(type, listener);
    }
    setMaxListeners(n) {
        return this;
    }

/**
 * Creates an instance of emitter. If `mapping` are provided they will automatically be passed into `on()` once construction is complete.
 * @class module:context.Emitter
 * @param {module:context.EventHandlers} [mapping] A mapping of event types to event listeners.
 * @mixes module:context.AsEventEmitter
 * @implements module:context.EventEmitter
 * @implements module:core.Destroyable
 * @implements module:core.Serializable
 * @see {@link https://github.com/nodejs/node/blob/master/lib/events.js}
 * @since 1.0.0
 * @example <caption>Using Emitter directly</caption>
 * const greeter = new Emitter();
 * greeter.on( 'hello', () => console.log( 'Hello!' ) );
 * greeter.emtest( 'hello' );
 * // Hello!
 * @example <caption>Extending Emitter using Classical inheritance</caption>
 * class Greeter extends Emitter {
 *  constructor(){
 *      super();
 *      this.on( 'greet', ( name ) => console.log( `Hello, ${ name }!` ) );
 *  }
 *
 *  greet( name ){
 *      this.emtest( 'greet', name );
 *  }
 * }
 *
 * const greeter = new Greeter();
 * greeter.greet( 'Jeff' );
 * // Hello, Jeff!
 * @example <caption>Extending Emitter using Prototypal inheritance</caption>
 * function Greeter(){
 *  Emitter.call( this );
 *  this.on( 'greet', ( name ) => console.log( `Hello, ${ name }!` ) );
 * }
 * Greeter.prototype = Object.create( Emitter.prototype );
 *
 * Greeter.prototype.greet = function( name ){
 *  this.emtest( 'greet', name );
 * };
 *
 * const greeter = new Greeter();
 * greeter.greet( 'Jeff' );
 * // Hello, Jeff!
 * @example <caption>Namespaced events</caption>
 * const greeter = new Emitter();
 * greeter.on( 'greeting:hello', ( name ) => console.log( `Hello, ${ name }!` ) );
 * greeter.on( 'greeting:hi', ( name ) => console.log( `Hi, ${ name }!` ) );
 * greeter.on( 'greeting', ( name ) => console.log( `${ name } was greeted.` );
 * greeter.emtest( 'greeting:hi', 'Mark' );
 * greeter.emtest( 'greeting:hello', 'Jeff' );
 * // Hi, Mark!
 * // Mark was greeted.
 * // Hello, Jeff!
 * // Jeff was greeted.
 * @example <caption>Predefined events</caption>
 * const greetings = {
 *      hello: function( name ){ console.log( `Hello, ${name}!` ),
 *      hi: function( name ){ console.log( `Hi, ${name}!` )
 *  },
 *  greeter = new Emitter( greetings );
 *
 * greeter.emtest( 'hello', 'Aaron' );
 * // Hello, Aaron!
 * @example <caption>One-time events</caption>
 * const greeter = new Emitter();
 * greeter.once( 'hello', ( name ) => console.log( `Hello, ${ name }!` ) );
 * greeter.emtest( 'hello', 'Jeff' );
 * greeter.emtest( 'hello', 'Terry' );
 * // Hello, Jeff!
 * @example <caption>Many-time events</caption>
 * const greeter = new Emitter();
 * greeter.many( 'hello', 2, ( name ) => console.log( `Hello, ${ name }!` ) );
 * greeter.emtest( 'hello', 'Jeff' );     // 1
 * greeter.emtest( 'hello', 'Terry' );    // 2
 * greeter.emtest( 'hello', 'Steve' );    // 3
 * // Hello, Jeff!
 * // Hello, Terry!
 */
export class Emitter extends AsEventEmitter(Null) {
    constructor(mapping) {
        super();
        defineProperty(this, 'destroyed', {
            value: false,
            configurable: true,
            enumerable: false,
            writable: false
        });
        typeof mapping !== 'undefined' && addEventHandlers(this, mapping, false);
    }
    /**
     * Remove all listeners, or those for the specified event `type`.
     * @function module:context.Emitter#clear
     * @param {module:context.EventType} [type] The event type.
     * @return {module:context.EventEmitter} The emitter.
     * @since 1.0.0
     * @example <caption>Clearing all event types</caption>
     * const greeter = new Emitter();
     * greeter.on( 'hello', () => console.log( 'Hello!' ) );
     * greeter.on( 'hi', () => console.log( 'Hi!' ) );
     * greeter.emtest( 'hello' );
     * // Hello!
     * greeter.emtest( 'hi' );
     * // Hi!
     * greeter.clear();
     * greeter.emtest( 'hello' );
     * greeter.emtest( 'hi' );
     * @example <caption>Clearing a specified event type</caption>
     * const greeter = new Emitter();
     * greeter.on( {
     *  'hello' : function(){ console.log( 'Hello!' ); },
     *  'hi'    : function(){ console.log( 'Hi!' ); }
     * } );
     * greeter.emtest( 'hello' );
     * // Hello!
     * greeter.emtest( 'hi' );
     * // Hi!
     * greeter.clear( 'hello' );
     * greeter.emtest( 'hello' );
     * greeter.emtest( 'hi' );
     * // Hi!
     */
    clear(type) {
        removeAllEventListeners(this, type);
        return this;
    }
    /**
     * @function module:context.Emitter#toJSON
     * @return {external:Object} An plain object representation of the emitter.
     * @since 1.3.0
     * @example
     * const greeter = new Emitter();
     * greeter.on( 'greet', ( name ) => console.log( `Hello, ${ name }!` ) );
     * greeter.on( 'greet', ( name ) => console.log( `Hi, ${ name }!` ) );
     *
     * console.log( greeter.toJSON() );
     * // { "listenerCount": { "greet": 2 } }
     *
     * greeter.destroy();
     *
     * console.log( greeter.toJSON() );
     * // "destroyed"
     */
    toJSON() {
        const json = createNull(), types = keys(this[$events]), length = types.length;
        let index = 0, type;
        json.listenerCount = createNull();
        for (; index < length; index++) {
            type = types[index];
            json.listenerCount[type] = this.listenerCount(type);
        }
        return json;
    }
    /**
     * @function module:context.Emitter#toString
     * @return {external:string} A string representation of the emitter.
     * @since 1.3.0
     * @example
     * const greeter = new Emitter();
     * greeter.on( 'greet', ( name ) => console.log( `Hello, ${ name }!` ) );
     * greeter.on( 'greet', ( name ) => console.log( `Hi, ${ name }!` ) );
     *
     * console.log( greeter.toString() );
     * // 'Emitter { "listenerCount": { "greet": 2 } }'
     *
     * greeter.destroy();
     *
     * console.log( greeter.toString() );
     * // 'Emitter "destroyed"'
     */
    toString() {
        return `${this[Symbol.toStringTag]} ${JSON.stringify(this.toJSON())}`.trim();
    }
    /**
     * Destroys the emitter.
     * @function module:context.Emitter#destroy
     * @since 1.0.0
     * @emits {:destroy} when the Emitter is destroyed
     */
    destroy() {
        if (!this.destroyed) {
            defineProperty(this, 'destroyed', {
                value: true,
                configurable: false,
                enumerable: false,
                writable: false
            });
            emitEvent(this, ':destroy', [], false);
            this.clear();
            this.toJSON = () => 'destroyed';
        }
    }
}
